اُن کی مہک نے دِل کے غُنچے کِھلا دئیے ہیں — Tazmeen 1

تَضْمِین (اَوَیس رَضا قادِری)
ــــــــــــــــــــــــــــــــــــــــــــــــــــ

کِس نے جہانِ کُن کے جلوے دِکھا دیے ہیں
سوئے ہوئے مُقدّر کِس نے جگا دیے ہیں
رنج و الم کے مارے سب مُسکُرا دیے ہیں
اُن کی مہک نے دِل کے غُنچے کِھلا دیے ہیں
جِس راہ چل گئے ہیں کُوچے بَسا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

قِصّہ غمِ جہاں کا جاکر کِسے سُنائیں
ہیں مُنتظِر کرم کی رُکتی ہوئی یہ سانسیں
کب رائیگاں گئی ہیں دِل سے اُٹھی جو آہیں
جب آگئی ہیں جوشِ رحمت پہ اُن کی آنکھیں
جلتے بُجھا دیے ہیں روتے ہنسا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

وہ مُبتلائے غم یا ویسے ہی رنج میں ہو
اُن کے غُلام ہو اور ایسے ہی رنج میں ہو
بھیجو دُرود اُن پر جیسے ہی رنج میں ہو
اُن کے نِثار کوئی کیسے ہی رنج میں ہو
جب یاد آگئے ہیں سب غم بُھلا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

جب حاضری کو گھر سے زائِر نِکلتے ہوں گے
جذباتِ قلبِ مُضطر اشکوں میں ڈھلتے ہوں گے
مُنہ مانگی آس پاکر کیسے مچلتے ہوں گے
ہم سے فقیر بھی اب پھیری کو اُٹھتے ہوں گے
اب تو غنی کے دَر پر بِستر جما دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

سرتاج عابِدوں کے ہیں فخر زاہِدوں کے
سُلطان دو جہاں کے مُختار مُمکِنوں کے
ھادی وہ دو جہاں کے مخزن وہ نعمتوں کے
اَسرا میں گُزرے جِس دم بیڑے پہ قُدسیوں کے
ہونے لگی سَلامی پرچَم جُھکا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

قُرآن لے چلا ہے ہم کو تُمہاری جانِب
لے آئے بارِ عِصیاں رو رو تُمہاری جانِب
بخشا گیا وہ جِس کا رُخ ہو تُمہاری جانِب
آنے دو یا ڈُبو دو اب تو تُمہاری جانِب
کَشتی تمہِیں پہ چھوڑی لنگر اُٹھا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

جُرم و خطا پر اپنے پردہ کِسی نے مانگا
دامانِ مُصطفیٰ ﷺ کا سایہ کِسی نے مانگا
جنّت میں سب سے اونچا رُتبہ کِسی نے مانگا
میرے کریم ﷺ سے گر قطرہ کِسی نے مانگا
دریا بہا دیے ہیں دُر بے بہا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

رب کی تجلیوں کی شیشہ کِسی نے مانگا
اُن سے مچل مچل کر جلوہ کِسی نے مانگا
زہرہ کے لاڈلوں کا صدقہ کِسی نے مانگا
میرے کریم ﷺ سے گر قطرہ کِسی نے مانگا
دریا بہا دیے ہیں دُر بے بہا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

مولی علی کا اُن سے جذبہ کِسی نے مانگا
اعداء پہ اپنے دائِم غلبہ کِسی نے مانگا
تسخیرِ دو جہاں کا نُسخہ کِسی نے مانگا
میرے کریم ﷺ سے گر قطرہ کِسی نے مانگا
دریا بہا دیے ہیں دُر بے بہا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

محشر میں جب پریشاں ہر ایک عبد ہوگا
پِھر آئے گا وہ جِس کو اُمّت کا درد ہوگا
سجدے میں سر کو رکھ کے پِھر محوِ حمد ہوگا
اللّٰہ کیا جہنّم اب بھی نہ سَرد ہوگا
رو رو کے مُصطفیٰ ﷺ نے دریا بہا دیے ہیں

༺❁━━━━━━━━━━━━━━━━━━━━━━━━❁༻

اُن کے سُخن کو حق نے کِتنا کیا مُکرّم
اشعار پر رضا کے تضمین کیا کہیں ہم
جو داغ کہہ گئے ہیں کہنا عُبید پیہم
مُلکِ سُخَن کی شاہی تُم کو رضاؔ مُسلَّم
جِس سَمْت آ گئے ہو سِکّے بِٹھا دیے ہیں
